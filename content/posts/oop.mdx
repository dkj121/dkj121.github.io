## 一、导入

OOP（Object-Oriented Programming，面向对象编程）是一种编程范式，它使用“对象”来设计应用程序和计算机程序。

很多人在初次**学习**OOP时，可能会感到有些困惑，因为传统的编程范式，如过程式编程，其实已经足以完成自己设想中大多数编程任务了。

毕竟，数据结构和算法是编程的核心，代码所做的一切，不过是对数据进行一步步的输入、处理和输出。

但是显然，当代码量从几十行增加到几百行、几千行甚至更多时，程序的复杂度也会随之增加，这时候仅仅只靠过程式编程的思路就显得力不从心了，这一步和下一步的关系变得越来越难以把握，代码的可读性和可维护性也会大幅下降。

而面向对象最重要的功能便体现在管理代码的层面。

## 二、猪脚饭！牛肉面！小糕点！

中国是有 56 个民族，坐拥五千年的历史文化，拥有丰富的饮食文化。中国的饮食文化博大精深，地域差异明显，形成了八大菜系和众多地方特色美食。

对于同一家的门头（餐馆）、不同的员工，他们的职责（下面称作 interface(接口) ）不同。对于同一个员工、不同的餐馆，他们的工作方法（下面称作 method(方法) ）也不同

但是对人来说，做饭、端盘子、整理物品确是手拿把掐的事情，我们或许可以称之为技能（下面称作 implement(实现) interface(接口) ），同样是做饭、端盘子、整理物品，不同的员工又有不同的技能实现（下面称作 polymorphism(多态)）

## 三、基础实现

```csharp
using System;

public class Restaurant

		public string Name { get; init; }
		public List<string> Menu { get; set; } = new List<string>();
		public List<Employee> Employees { get; set; } = new List<Employee>();

										return order;
								}
						}
				}
				else
				{
						Console.WriteLine($"Sorry, {foodItem} is not available in the menu.");
						return null;
				}
		}
```

```csharp
using System;

/// <summary>
/// 骑手类，包含骑手的基本信息，如姓名和订单列表。

public class DeliveryDriver
{
		public string Name { get; init; }
		public List<Order> Orders { get; set; } = new List<Order>();

		public DeliveryDriver(string name)
		{
				Name = name;
		}

		public void AddOrder(Order order)
		{
				if (order is null)
				{
						Console.WriteLine("Cannot add a null order.");
						return;
				}
				Orders.Add(order);
		}

		/// <summary>
		/// 模拟骑手送餐的过程，遍历订单列表并打印送餐信息，送完后清空订单列表。
		/// </summary>
		public void DeliverOrders()
		{
				foreach (var order in Orders)
				{
						Console.WriteLine($"Delivering order {order.Origin} to {order.Destination} with food item {order.FoodItem}");
						order.IsDelivered = true;
				}
				Orders.Clear();
		}
```

在这里，我们通过最简单的对象来实现了一个外卖服务系统的基本功能。每个类都代表了系统中的一个实体，并且通过方法和属性来定义它们的行为和状态。

但是，随着系统的复杂度增加，我们可能需要引入更多的类和接口来管理不同的功能和行为，这就是面向对象编程的魅力所在。

例如，对于当前员工对象来说，一个员工只能拥有一个技能，但是出于资本家的善意，我们希望员工可以拥有多项技能，这样他们就可以在不同的岗位上工作。

最简单的实现就是在 Employee 类中添加一个技能列表，并在创建员工对象时指定其技能。然后，我们可以通过检查员工的技能列表来判断他们是否具备某项技能。

例如：

但是，这种实现方式不论是性能上，还是实现上都并不优雅（需要大量的判断以及搜索），也不符合面向对象编程的原则。我们可以通过引入接口（interface）来更好地管理员工的技能。

## 四、接口(interface)

例如，考虑到并不是每一个员工都会做饭，我们定义一个 ICookable 接口，表示具备做饭技能的员工。然后，我们可以让具备做饭技能的员工类实现这个接口。

```csharp
/// ICookable.cs
namespace Food;

/// <summary>
/// ICookable 接口，表示具备做饭技能的员工。
/// </summary>
public interface ICookable
{
		void Cook(string foodItem);
}
```

```csharp
/// Employee.cs
namespace Food;

/// <summary>
/// 普通员工类，表示普通员工
/// </summary>
public abstract class EmployeeBase
{
	public abstract string Name { get; init; }
	public abstract bool IsAvailable { get; set; }

	public Employee(string name)
	{
		Name = name;
	}
}

/// <summary>
/// 厨师类，表示具备做饭技能的员工
/// </summary>
public class Chef : EmployeeBase, ICookable
{
	public override string Name { get; init; }
	public override bool IsAvailable { get; set; } = true;

	public Chef(string name) : base(name)
	{
		Name = name;
	}
	public void Cook(string foodItem)
	{
		Console.WriteLine($"Cook {Name} is cooking {foodItem}");
	}
}
```

```csharp
/// <summary>
/// 餐馆类，包含餐馆的基本信息，如名称、地址和菜单。
/// </summary>
public class Restaurant
{
	public string Name { get; init; }
	public List<string> Menu { get; set; } = new List<string>();
	public List<EmployeeBase> Employees { get; set; } = new List<EmployeeBase>();

	public Restaurant(string name)
	{
		Name = name;
	}

	/// <summary>
	/// 创建订单的方法，检查菜单中是否包含所点的食物，如果包含，则调用厨师的 Cook 方法进行烹饪，并返回订单对象；否则，提示食物不在菜单中。
	/// </summary>
	/// <param name="foodItem">所点的食物名称</param>
	/// <param name="customer">客人对象</param>
	public Order CreateOrder(string foodItem, Customer customer)
	{
		if (Menu.Contains(foodItem))
		{
			var cook = from employee in Employees
								 where employee is ICookable && employee.IsAvailable
								 select employee as ICookable;
			if (cook == null)
			{
				Console.WriteLine("No available cook found.");
				return null;
			}
			cook.First().Cook(foodItem);
			Console.WriteLine($"Order created: {foodItem} from {Name} to {customer.Name}");
			return new Order(Name, customer.Address, foodItem);
		}
		else
		{
			Console.WriteLine($"Sorry, {foodItem} is not available in the menu.");
			return null;
		}
	}
}
```

```csharp
/// Restaurant.cs
namespace Food;

/// <summary>
/// RestaurantBase 抽象类，表示餐馆的基本信息和行为。
/// </summary>
public abstract class RestaurantBase
{
	public abstract string Name { get; init; }
	public virtual List<string> FoodMenu { get; set; }
	public virtual List<string> DrinkMenu { get; set; }
	public abstract List<EmployeeBase> Employees { get; set; }

	public RestaurantBase(string name)
	{
		Name = name;
	}

	public abstract void CleanRestaurant();
}

/// <summary>
/// 食物餐馆类，表示提供食物的餐馆。
/// </summary>
public class FoodRestaurant : RestaurantBase, IEatable
{
	public override string Name { get; init; }
	public override List<string> FoodMenu { get; set; } = new List<string>();
	public override List<EmployeeBase> Employees { get; set; } = new List<EmployeeBase>();

	public FoodRestaurant(string name) : base(name)
	{
		Name = name;
	}

	public override void CleanRestaurant()
	{
		var cleaner = from employee in Employees
									where employee is ICleanable && employee.IsAvailable
									select employee as ICleanable;
		if (cleaner == null)
		{
			Console.WriteLine("No available cleaner found.");
			return;
		}
		cleaner.First().Clean();
		Console.WriteLine($"Restaurant {Name} has been cleaned.");
	}

	public Order CreateEatOrder(string foodItem, Customer customer)
	{
		if (FoodMenu.Contains(foodItem))
		{
			var cook = from employee in Employees
								 where employee is ICookable && employee.IsAvailable
								 select employee as ICookable;
			if (cook == null)
			{
				Console.WriteLine("No available cook found.");
				return null;
			}
			cook.First().Cook(foodItem);

			var waiter = from employee in Employees
									 where employee is IWaitable && employee.IsAvailable
									 select employee as IWaitable;
			if (waiter == null)
			{
				Console.WriteLine("No available waiter found.");
				return null;
			}
			waiter.First().Serve(foodItem);
			Console.WriteLine($"Order created: {foodItem} from {Name} to {customer.Name}");
			return new Order(Name, customer.Address, foodItem);
		}
		else
		{
			Console.WriteLine($"Sorry, {foodItem} is not available in the menu.");
			return null;
		}
	}
}

/// <summary>
```

```csharp
/// <summary>
/// 食物餐馆类，表示提供食物的餐馆。
/// </summary>
public class FoodRestaurant : RestaurantBase, IEatable
{
	public override string Name { get; init; }
	public override List<string> FoodMenu { get; set; } = new List<string>();
	public override List<EmployeeBase> Employees { get; set; } = new List<EmployeeBase>();

	public FoodRestaurant(string name) : base(name)
	{
		Name = name;
	}

	public override void CleanRestaurant()
	{
		var cleaner = from employee in Employees
									where employee is ICleanable && employee.IsAvailable
									select employee as ICleanable;
		if (cleaner == null)
		{
			Console.WriteLine("No available cleaner found.");
			return;
		}
		cleaner.First().Clean();
		Console.WriteLine($"Restaurant {Name} has been cleaned.");
	}

	public Order CreateEatOrder(string foodItem, Customer customer)
	{
		if (FoodMenu.Contains(foodItem))
		{
			var cook = from employee in Employees
								 where employee is ICookable && employee.IsAvailable
								 select employee as ICookable;
			if (cook == null)
			{
				Console.WriteLine("No available cook found.");
				return null;
			}
			cook.First().Cook(foodItem);

			var waiter = from employee in Employees
									 where employee is IWaitable && employee.IsAvailable
									 select employee as IWaitable;
			if (waiter == null)
			{
				Console.WriteLine("No available waiter found.");
				return null;
			}
			waiter.First().Serve(foodItem);
			Console.WriteLine($"Order created: {foodItem} from {Name} to {customer.Name}");
			return new Order(Name, customer.Address, foodItem);
		}
		else
		{
			Console.WriteLine($"Sorry, {foodItem} is not available in the menu.");
			return null;
		}
	}
}

/// <summary>

/// </summary>
public class DeliveryDriver
{
	public string Name { get; init; }
	public List<Order> Orders { get; set; } = new List<Order>();

	public DeliveryDriver(string name)
	{
		Name = name;
	}

	/// <summary>
	/// 模拟骑手送餐的过程，遍历订单列表并打印送餐信息，送完后清空订单列表。
	/// </summary>
	public void DeliverOrders()
	{
		foreach (var order in Orders)
		{
			Console.WriteLine($"Delivering order {order.Origin} to {order.Destination} with food item {order.FoodItem}");
			order.IsDelivered = true;
		}
		Orders.Clear();
	}
}

/// <summary>

	{
		Name = name;
		Address = address;
	}

	{
		if (!order.IsDelivered)
		{

			return;
		}
		Console.WriteLine($"Customer {order.Origin} received order from {order.Destination} with food item {order.FoodItem}");
	}
}
```
